    Usage

    One element, every framework.

    HTML / CDN

    <script type="module"
  src="https://cdn.jsdelivr.net/npm/@lumi-icons/core/+esm"></script>

<lumi-icon name="rocket" animation="launch"></lumi-icon>

    React

    npm i @lumi-icons/core @lumi-icons/react

import { LumiIcon } from "@lumi-icons/react";

<LumiIcon name="rocket" size={40} />

    Vue

    npm i @lumi-icons/core @lumi-icons/vue

import { LumiIcon } from "@lumi-icons/vue";

<LumiIcon name="rocket" :size="40" />
    AttributeValuesDefault
